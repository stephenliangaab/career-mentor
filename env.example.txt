# 职径导航 - 环境变量配置模板
# 复制此文件的内容到Vercel项目的Environment Variables中

# ==================== OpenAI配置 ====================
# 获取方式: https://platform.openai.com/api-keys
# 用途: 生成AI推荐结果
OPENAI_API_KEY=sk-your-openai-api-key-here

# ==================== 兼容 DeepSeek / 其他 OpenAI 兼容服务(可选) ====================
# 说明：本项目后端支持两种模式：
# 1) OpenAI SDK（默认）：使用 OPENAI_API_KEY
# 2) DeepSeek 原生接口（推荐你这样配）：使用 DEEPSEEK_API_KEY，并指定 AI_PROVIDER=deepseek
#
# DeepSeek（按你提供的官方 cURL）：
# AI_PROVIDER=deepseek
# DEEPSEEK_API_KEY=sk-or-deepseek-key-here
# DEEPSEEK_BASE_URL=https://api.deepseek.com
# AI_MODEL=deepseek-chat
#
# 你也可以用 AI_API_KEY 单独存 Key（不想沿用 OPENAI_API_KEY 这个名字时用）
# AI_API_KEY=sk-or-deepseek-key-here

# ==================== Vercel KV存储配置 ====================
# Vercel会自动注入以下环境变量,无需手动配置
# KV_REST_API_URL=
# KV_REST_API_TOKEN=
# KV_REST_API_READ_ONLY_TOKEN=

# ==================== 限流配置(可选) ====================
# 如需自定义限流参数,可添加以下配置
# MONTHLY_LIMIT=1000
# DAILY_LIMIT=50
# IP_DAILY_LIMIT=5

# ==================== 其他配置 ====================
# 生产环境URL(用于CORS等)
# PRODUCTION_URL=https://your-domain.com
